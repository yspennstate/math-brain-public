    [Theorem thm:dim_free_unbounded of arXiv:2411.07878, 'Bernstein-type and Bennett-type inequalities for unbounded matrix martingales'] Let $\mathbf{X}_1, \dots, \mathbf{X}_n \in \mathbb{H}(d)$ be a martingale difference sequence adapted to a filtration $(\mathscr{F}_i)^n_{i=0}$ with $\mathscr{F}_0 = \{\varOmega, \varnothing\}$. Fix $\alpha > 0$ and set for all $i \in [n]$ \[ \boldsymbol{\Sigma}_i \coloneqq \E\lrs{\mathbf{X}^2_i \middle| \mathscr{F}_{i-1}}, \quad U_i \coloneqq \norm*{\lambda_{\max}(\mathbf{X}_i)_+ \big| \mathscr{F}_{i-1}}_{\psi_{\alpha}}. \] Fix $\boldsymbol{\Sigma} \succcurlyeq 0$, $U \ge K > 0$ and define the event \begin{equation} E \coloneqq \lrc{\sum^n_{i = 1} \boldsymbol{\Sigma}_i \preccurlyeq \boldsymbol{\Sigma}, \quad \sum^n_{i = 1} U^{2}_i \le U^2, \quad \max_{i \in [n]} U_i \le K}. \end{equation} Let $\sigma^2 \coloneqq \lambda_{\max}(\boldsymbol{\Sigma})$, and $z$ be as in (equation eq:def_z of the paper). Then for any $\mathrm{x} > 0$ with probability at least $\mathbb{P}(E) - (e\cdot r(\boldsymbol{\Sigma}) +1) e^{-\mathrm{x}} - e^{-\mathrm{x}} \ind[\alpha < 1]$, it holds for any $0 < \varepsilon\le 1$ \begin{align} \max_{k \in [n]} \lambda_{\max}(\mathbf{S}_k) \le& (1 + \varepsilon)\sigma \sqrt{2\mathrm{x}} + \frac{7 \lr{\ln \frac{8}{\varepsilon}}^{1/\alpha} Kz \mathrm{x}}{\min\lrc{2 \alpha z^{\alpha}, \hspace{0.2em}\underline{\log}\hspace{0.2em}\lr{\lr{\frac{K z}{\sigma}}^2 \mathrm{x}} }} \nonumber \\ &+ \frac{3 K}{\alpha} \left(2 \mathrm{x} + 2\ln\lr{\frac{4 U}{K}} + \frac{4}{\alpha}\ln \lr{\frac{4}{\alpha e}} \right)^{\frac{1-\alpha}{\alpha}} \ind[\alpha < 1] , \end{align} and moreover, \begin{align} \max_{k \in [n]} \lambda_{\max}(\mathbf{S}_k) \le& (1 + \varepsilon)\sigma \sqrt{2 \mathrm{x}} + 2 \lr{\ln \frac{8}{\varepsilon}}^{1/\alpha} Kz \mathrm{x} \nonumber \\ &+ \frac{3 K}{\alpha} \left(2 \mathrm{x} + 2\ln\lr{\frac{4 U}{K}} + \frac{4}{\alpha}\ln \lr{\frac{4}{\alpha e}} \right)^{\frac{1-\alpha}{\alpha}} \ind[\alpha < 1] . \end{align} (Proof in the paper; no proof environment follows the statement in the TeX.)
