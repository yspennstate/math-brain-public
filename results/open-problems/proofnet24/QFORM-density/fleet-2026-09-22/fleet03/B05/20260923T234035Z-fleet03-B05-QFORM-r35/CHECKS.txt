G=[[g11,g12],[g12,g22]],
K=[[1,0,0,0],[a11,a12,1,0],[a21,a22,0,1]],
M=diag-block(G,y2,z2), F=K M K^T.

  a12^2*(g11*g22-g12^2)*z2
+ a22^2*(g11*g22-g12^2)*y2
