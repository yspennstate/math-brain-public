\begin{abstract}
We assemble the uniform-cube reduction C005--C006--C018, the
C019 stopping-tree criterion, and A097's compatible transfer of complete
terminal references. All proofs needed for these implications are included.
The root and singleton-to-pair erasure layers are paid; the
dimension-independent higher-layer payment is an explicit conjecture.
The B05 Boolean obstruction excludes only a specified all-coordinate local
refund. Full FEI remains open. We assemble existing arguments and refine
low-total-influence completeness to equality of the optimal erasure constants.
Independent full-chain reviews by A090 and A094 have not been received.
Historical C024 is kept separate from the
finite-tree transfer, and no biased-measure, percolation or threshold theorem
is silently imported.
\end{abstract}

\section{Conventions and mathematical scope}
For every finite $n\ge0$, $f:\{-1,1\}^n\to\{-1,1\}$ is Boolean,
$a_S=\E f\chi_S$, $P_f(S)=p_S=a_S^2$,
$H(f)=\sum_Sp_S\log_2(1/p_S)$, and
$I(f)=\sum_S|S|p_S=\sum_i d_i(f)$, where
$d_i(f)=\Pr(f(x)\ne f(x^{\oplus i}))$.
Every unmarked entropy, cross-entropy and divergence is in bits;
$c_0=2/\ln2$. Zero entropy atoms contribute zero. Spectral conditioning
does not mean physical restriction. We write
\[
 B_i(f)=H(S_i\mid S_{-i}),\quad B(f)=\sum_i B_i(f),\quad
 \mathcal L(f)=\sum_{S:p_S>0}p_S\sum_{i\in S}
                    \log_2(1+p_{S\setminus\{i\}}/p_S).
\]
The symbol $\mathcal L$ is the aggregate C018 downward charge. B05's
quantity named $L$ is instead a single $B_i$; Section~\ref{sec:b05}
calls it $B_y$ to prevent an erroneous substitution.
\[
C_{\rm FEI}=\sup_{n,f:I(f)>0} H(f)/I(f),\quad
K_E=\sup_{n,f:I(f)>0} B(f)/I(f),\quad
K_{\mathcal L}=\sup_{n,f:I(f)>0}\mathcal L(f)/I(f).
\]
These suprema may be infinite and need not be attained.

\section{C005: actual singleton restrictions}
\begin{lemma}[Entropy and influence identities, source ER Sections 1--4]
\label{lem:single}
For every Boolean $f$ and coordinate $i$, write $f_\pm=f_{i=\pm1}$.
Then
\[
H(f)-\tfrac12(H(f_+)+H(f_-))=B_i(f)+J_i(f),\qquad
I(f)=d_i(f)+\tfrac12(I(f_+)+I(f_-)),
\]
where $J_i$ is the Jensen--Shannon divergence of $P_{f_+},P_{f_-}$,
and
\[
0\le J_i\le c_0\sum_{T\not\ni i}\min(p_T,p_{T\cup\{i\}})
             \le c_0\min(d_i,1-d_i).
\]
\end{lemma}
\begin{proof}
Set $u=a_T$, $v=a_{T\cup\{i\}}$, $m=u^2+v^2$.
The two actual coefficients are $u+v,u-v$ and their mean squared
mass is $m$. The chain rule gives
$B_i=\sum_T[h(u^2)+h(v^2)-h(m)]$, where $h(t)=-t\log_2t$.
Meanwhile
$J_i=\sum_T[h(m)-\tfrac12h((u+v)^2)-\tfrac12h((u-v)^2)]$.
Adding cancels $h(m)$ and proves the identity. The child laws are
normalized by Booleanity and Parseval, so $J_i\ge0$ by concavity.

For $0\le z\le1$, let
$D(z)=\tfrac12[(1+z)\ln(1+z)+(1-z)\ln(1-z)]$.
Then $D(z)\le1-\sqrt{1-z^2}$: both vanish at zero;
their derivatives are $\operatorname{atanh}z$ and
$z/\sqrt{1-z^2}$, and the derivative of the difference of these
derivatives is $(1-z^2)^{-3/2}-(1-z^2)^{-1}\ge0$.
Continuity handles $z=1$.
At $m>0$ the JS contribution is $mD(2|uv|/m)/\ln2$,
at most $(m-|u^2-v^2|)/\ln2=c_0\min(u^2,v^2)$.
Zero fibers vanish. Sum over $T$ and use
$\sum_Tv^2=d_i$, $\sum_Tu^2=1-d_i$.
For $j\ne i$, averaging the $j$-pivotal indicator over the fixed
$i$ coordinate gives $d_j(f)=\tfrac12(d_j(f_+)+d_j(f_-))$.
Summation proves the influence identity.
\end{proof}

\section{C006: actual blocks and their channel price}
\begin{lemma}[Cube log-Sobolev inequality in the required normalization]
\label{lem:lsi}
For $g\ge0$ on a finite uniform cube,
\[
 \Ent(g^2):=\E[g^2\ln(g^2/\E g^2)]
 \le\tfrac12\sum_i\E(g(x)-g(x^{\oplus i}))^2.
\]
\end{lemma}
\begin{proof}
On one bit, by homogeneity the required inequality for values $t,1$
is $F(t)\ge0$, where
\[
F(t)=(t^2+1)\ln((t^2+1)/2)-2t^2\ln t+(t-1)^2.
\]
Here $F(1)=0$ and
$F'(t)=2t\ln((t^2+1)/(2t^2))+2(t-1)$.
For $t\ge1$, the difference
$1-1/t-\ln(2t^2/(t^2+1))$ is zero at 1 and has derivative
$(t-1)^2/[t^2(t^2+1)]\ge0$, proving $F'\ge0$.
The identity $F(t)=t^2F(1/t)$ handles $t<1$, and continuity
handles zero values.

For completeness, entropy tensorization follows as follows. For a
nonnegative $w$ with mean $m>0$, put $R(x)=w(x)2^{-n}/m$.
Then $\Ent(w)/m=D_{\rm nat}(R\|U)$.
The divergence chain rule writes this as the sum of the divergences
of $R(X_i\mid X_{<i})$ from the uniform one-bit law.
Replacing $X_{<i}$ by $X_{-i}$ increases each mean divergence,
because conditioning decreases entropy. Direct substitution of the
density shows that the resulting $i$th term, multiplied by $m$, is
$\E_{U,-i}\Ent_{U,i}(w)$.
Thus $\Ent(w)\le\sum_i\E_{-i}\Ent_i(w)$.
Apply the one-bit bound to $w=g^2$ on each fiber. The case $m=0$
is immediate.
\end{proof}

\begin{lemma}[Block identity and paid channel, source BR Sections 1--3]
\label{lem:block}
Let $A$ be a nonempty physical block, $C=[n]\setminus A$,
$X$ a uniform assignment on $A$, and conditionally on $X=x$ let
$T$ have the spectral law of $f_x$. Then
\[
H(f)-\E_xH(f_x)=B_A(f)+J_A(f),\quad
B_A(f)=H(S_A\mid S_C),\quad J_A(f)=I(X;T),
\]
\[
0\le J_A(f)\le c_0 I_A(f),\qquad
I(f)=I_A(f)+\E_xI(f_x),\quad I_A=\sum_{i\in A}d_i.
\]
\end{lemma}
\begin{proof}
Let $v_T(x)=\E_y f(x,y)\chi_T(y)$,
$P_x(T)=v_T(x)^2$, and $q(T)=\E_xP_x(T)$.
Parseval in $x$ gives $q(T)=\sum_{U\subseteq A}a_{U\cup T}^2$,
the actual $C$-marginal. Hence
$H(f)=H(q)+H(S_A\mid S_C)$ and
$J_A=H(q)-\E_xH(P_x)$, proving the first identity.
For $q(T)>0$, apply Lemma~\ref{lem:lsi} to
$g_T(x)=|v_T(x)|/\sqrt{q(T)}$, multiply by $q(T)$ and sum.
The left side is $(\ln2)J_A$; the right side is at most
\[
\tfrac12\sum_{i\in A}\E_x\sum_T(v_T(x)-v_T(x^{\oplus i}))^2
=\tfrac12\sum_{i\in A}\E_{x,y}(f(x,y)-f(x^{\oplus i},y))^2
=2I_A(f).
\]
The middle equality is Parseval on $C$; the last uses Boolean
values. Zero $q$ fibers vanish. Averaging each pivotal indicator
on $C$ over $X$ proves influence conservation.
\end{proof}

\begin{lemma}[B06 R1: block-to-singleton, one canonical source ID]
\label{lem:r1}
For any law on subsets and any block $A$,
\[
H(S_A\mid S_{A^c})\ge\sum_{i\in A}H(S_i\mid S_{-i}).
\]
For Boolean $f$ with $I_A(f)>0$, a bound $B_A(f)\le KI_A(f)$
therefore gives some relevant $i\in A$ with $B_i(f)\le Kd_i(f)$.
A bound $\Delta_AH(f)\le KI_A(f)$ gives some such $i$ with
$\Delta_iH(f)\le(K+c_0)d_i(f)$.
\end{lemma}
\begin{proof}
Order $A$ and expand the block conditional entropy by the chain
rule. Each term conditions only on $A^c$ and preceding bits of
$A$; conditioning further on all other bits can only reduce it.
Sum, then apply influence-weighted averaging. For the final
statement $B_A\le\Delta_AH$ by Lemma~\ref{lem:block}, and
$\Delta_iH\le B_i+c_0d_i$ by Lemma~\ref{lem:single}.
Irrelevant coordinates have zero $B_i$.
\end{proof}
This is B06-LEAD-R1-BLOCK-TO-SINGLETON version 1, reproduced here.
A block cannot rescue the same node if every singleton erasure
rate fails; its distinct role is skipping intermediate restrictions.

\section{C018: the proved equivalence and its open Boolean leaf}
\begin{lemma}[Erasure comparison and downward charge, ER Sections 3--6]
\label{lem:downward}
For every Boolean $f$,
\[
\mathcal L(f)\le B(f)\le\mathcal L(f)+I(f)/\ln2,\qquad B(f)\le H(f).
\]
The coefficients also satisfy
\[
\sum_Sa_S^2=1,\qquad \sum_Sa_Sa_{S\triangle U}=0
\quad(\varnothing\ne U\subseteq[n]).
\]
Conversely these identities for real $a$ imply its inverse Walsh
transform is Boolean.
\end{lemma}
\begin{proof}
For nonnegative $r,s$,
$h(r)+h(s)-h(r+s)=r\log_2(1+s/r)+s\log_2(1+r/s)$,
with continuous endpoint conventions. On each edge $(T,T\cup\{i\})$
the second term is the downward charge. The first is at most
$s/\ln2$, by $\ln(1+t)\le t$, including zero endpoints by continuity.
Sum to prove the sandwich. In any coordinate order,
$H(S_i\mid S_{-i})\le H(S_i\mid S_{<i})$; the entropy chain
rule proves $B\le H$. Fourier transforming $f^2=1$ gives the
convolution identities. Conversely they say all nonconstant
coefficients of the square of the inverse transform vanish and its
constant coefficient is 1, so its square is pointwise 1.
\end{proof}

\begin{theorem}[C018; implication proved, payment open]
\label{thm:erasure}
In the extended real sense,
\[
K_E\le C_{\rm FEI}\le K_E+c_0,\qquad
K_{\mathcal L}\le K_E\le K_{\mathcal L}+1/\ln2.
\]
In particular any uniform bound $B(f)\le KI(f)$ implies
$H(f)\le(K+c_0)I(f)$; any uniform bound
$\mathcal L(f)\le KI(f)$ implies
$H(f)\le(K+3/\ln2)I(f)$.
\end{theorem}
\begin{proof}
The lower bound follows from Lemma~\ref{lem:downward}.
Assume $K_E<\infty$. Summing Lemma~\ref{lem:single} gives
$\sum_i\Delta_iH\le(K_E+c_0)I(f)$.
Irrelevant coordinates have $B_i=J_i=\Delta_iH=0$.
If $f$ is nonconstant, some relevant coordinate satisfies
$\Delta_iH\le(K_E+c_0)d_i$ by weighted averaging.
Induct on dimension, restricting that coordinate, and use the exact
influence identity to obtain $H(f)\le(K_E+c_0)I(f)$.
Constant functions have $H=I=0$. If $K_E=\infty$ the upper
comparison is vacuous. The remaining comparisons and constant follow
from the sandwich.
\end{proof}

\section{Paid universal layers and the exact remaining erasure}
Put $\phi(r,s)=h(r)+h(s)-h(r+s)$ and define
\[
B_0=\sum_i\phi(a_\varnothing^2,a_{\{i\}}^2),\quad
T_{12}=\sum_i\sum_{j\ne i}\phi(a_{\{i\}}^2,a_{\{i,j\}}^2),\quad
R_{\ge2}=\sum_{|S|\ge2}\sum_{i\notin S}\phi(a_S^2,a_{S\cup\{i\}}^2).
\]
Thus $B=B_0+T_{12}+R_{\ge2}$ exactly, with each cube edge
counted once from its lower set.
\begin{lemma}[B20 root-pair payment, version 1]
\label{lem:root}
For every Boolean $f$, $B_0\le c_r I(f)$, where
$c_r=2/(\sqrt e\ln2)$.
\end{lemma}
\begin{proof}
For $r+s\le1$, $\phi(r,s)\le s\log_2(e/s)$, by
$r\ln(1+s/r)\le s$ and $\ln((r+s)/s)\le\ln(1/s)$.
The physical derivative identity gives $|a_{\{i\}}|\le d_i(f)$.
The function $s\log_2(e/s)$ is increasing on $(0,1]$, so its
value at $a_{\{i\}}^2$ is at most $d_i^2\log_2(e/d_i^2)$.
Finally $\sup_{0<t\le1}t\log_2(e/t^2)=2/(\sqrt e\ln2)$,
as differentiation shows; sum over $i$.
\end{proof}

\begin{theorem}[A096 universal layer payment, version 1]
\label{thm:level12}
For every Boolean $f$, writing
$W_2=\sum_{i<j}a_{\{i,j\}}^2$ and
$H_2=\sum_{i<j}a_{\{i,j\}}^2\log_2(1/a_{\{i,j\}}^2)$,
\[
H_2\le\frac{155600}{50301}I(f),\qquad
T_{12}\le c_{12}I(f),\quad c_{12}=\frac{16700}{2187}<8.
\]
\end{theorem}
\begin{proof}
Let $\epsilon_i=d_i(f)$ and
$h_i=(f_{i=+1}-f_{i=-1})/2$. Booleanity implies
$h_i\in\{0,\pm1\}$ and $\Pr(h_i\ne0)=\E h_i^2=\epsilon_i$.
For $L_i=\sum_{j\ne i}a_{\{i,j\}}x_j$, let
$u_i=\E L_i^2=\E h_iL_i=\sum_{j\ne i}a_{\{i,j\}}^2$.
Independence and $\cosh v\le e^{v^2/2}$ give
$\E e^{t|L_i|}\le2e^{t^2u_i/2}$.
The scalar bound follows by integrating $\tanh v\le v$.
Conditioning on $h_i\ne0$ and applying Jensen gives
\[
u_i\le\epsilon_i\left(\frac{\ln(2/\epsilon_i)}t+\frac{tu_i}2\right),
\quad\text{hence}\quad
u_i\le2\epsilon_i^2\ln(2/\epsilon_i)
\]
after optimization; zero support or row energy is immediate.

For $I,W_2>0$, the unordered-pair reference
$Q(\{i,j\})=2\epsilon_i\epsilon_j/I^2$ has total mass at most
one and is positive on all occupied pairs. Gibbs for the normalized
level law $a_{\{i,j\}}^2/W_2$, retaining its level mass, gives
\[
H_2\le\sum_i u_i\log_2(1/\epsilon_i)
                 +W_2\log_2(I^2/(2W_2)).
\]
Using $t=-\ln\epsilon_i$ and the maxima
$\sup t^2e^{-t}=4/e^2$, $\sup te^{-t}=1/e$, the first sum is
at most $[8/(e^2\ln2)+2/e]I$.
For every $z>0$, $\ln z\le2\sqrt z/e$, so the second term
is at most $\sqrt2 I\sqrt{W_2}/(e\ln2)\le\sqrt2 I/(e\ln2)$.
This includes negative logarithms. Thus $H_2\le K_2I$ for
$K_2=8/(e^2\ln2)+2/e+\sqrt2/(e\ln2)$.
Since every edge has $r+s\le1$, its cost is at most
$s\log_2(1/s)+s/\ln2$. Each pair is counted twice and
$2W_2\le I$, giving $T_{12}\le(2K_2+1/\ln2)I$.
The rational comparisons $e>27/10$, $\ln2>69/100$ and
$\sqrt2<71/50$ follow respectively from the exponential series
through degree five, the first two positive atanh terms at $1/3$,
and squaring. Substitution yields
$K_2<155600/50301$ and $2K_2+1/\ln2<16700/2187$.
Constants or $W_2=0$ give zero contributions.
\end{proof}
The canonical ID is A096-UNIVERSAL-LEVEL12-ERASURE version 1.
A098 independently accepted precisely this all-dimensional layer;
that review does not pay the sum of all higher levels.

\section{B20 low-total-influence completeness, strengthened version 2}
\label{sec:low}
For finite-valued $s:\mathbb N\to[0,\infty)$ with $s(N)=o(N)$,
let $K_E(s)$ be the supremum $B(F)/I(F)$ over every dimension
$N$ and every Boolean $F$ with $0<I(F)\le2^{-s(N)}$.
\begin{theorem}[B20-ERASURE-LOW-INFLUENCE-COMPLETE, proposed version 2]
\label{thm:low}
In the extended reals, $K_E(s)=K_E$. Consequently
$C_{\rm FEI}\le K_E(s)+2/\ln2$.
\end{theorem}
\begin{proof}
Disjoint products have product Fourier laws; conditional entropies
and physical influence add. For a nonconstant seed $f$ on $n\ge1$
bits take $m$ copies and a fresh fair sign $z$, and put
$h_m=z\prod_{j=1}^m f(x^{(j)})$.
Then $\E h_m=0$, $B(h_m)=mB(f)$ and $I(h_m)=mI(f)+1$.
The $z$-frequency is deterministic and its erasure is zero.

Activate $h_m$ at one point of $k\ge1$ controls: $F=1+A_k(h_m-1)$,
$\delta=2^{-k}$. The constant coefficient is $1-\delta$;
every nonempty core coefficient is $\delta\widehat h_m(S)$
in every control-frequency slice; empty-core/nonempty-control
coefficients have magnitude $\delta$. Partition the core-coordinate
edges by their residual core index. With $b_i=\widehat h_m(\{i\})^2$,
the original root packet is $\phi(0,b_i)=0$ and
\[
B_{\rm data}(F)=\delta B(h_m)
 +\delta(1-\delta)\sum_i\phi(1,b_i)
 +\sum_i\phi((1-\delta)^2,\delta^2b_i)\ge\delta mB(f).
\]
Control erasures are nonnegative. Direct physical edge counts give
$I(F)=\delta[mI(f)+1+k]$, since $h_m$ is balanced.

Set $\bar s(N)=\max_{M\le N}s(M)$; it is nondecreasing and
$o(N)$, because the finite initial maximum is $o(N)$ and the
tail is bounded by $\varepsilon N$ for every fixed $\varepsilon>0$.
Define $k_m=\lceil\bar s(2mn)+\log_2(3mn)\rceil+1$ and
$N_m=mn+1+k_m$. For large $m$, $k_m\le mn-1$, $N_m\le2mn$, and
\[
0<I(F_m)\le2^{-k_m}3mn
\le2^{-\bar s(2mn)}\le2^{-\bar s(N_m)}.
\]
If $K=K_E(\bar s)$ is finite, admissibility implies
$mB(f)\le K[mI(f)+1+k_m]$.
Divide by $m$ and let $m\to\infty$; $k_m/m\to0$ gives
$B(f)\le KI(f)$. Hence
$K_E\le K_E(\bar s)\le K_E(s)\le K_E$.
If $K_E=\infty$, the same argument excludes finite $K_E(s)$;
the extended-real equality follows. C018 proves the corollary.
\end{proof}
This refines the source's valid additive bound
$K_E\le K_E(s)+c_r$; the original version is not refuted.
The fresh balancing makes the root-pair correction zero, while its
one unit of influence disappears after tensor amplification.
The proposed version has a complete proof and exact finite identity
validation, but independent reviews A090/A094 are pending.
This is aggregate erasure transport. It makes no same-constant
claim about a \emph{selected} data-coordinate witness; A005's
control-witness counterexample therefore remains applicable.
Finiteness of the low-influence constant is still unproved.

\section{C019: finite adaptive stopping trees and one influence budget}
At an internal node $h$, let $f_h$ be the actual remaining Boolean
function. Choose a nonempty block $A_h$ using only the physical
history. Query it uniformly; its child assignments $x$ each have
conditional probability $\alpha_h=2^{-|A_h|}$.
No previously fixed coordinate may be queried along the same path.
The tree is finite; it may stop before reaching constants.
Let $\pi_h$ be its actual reach probability and put
\[
B_h=B_{A_h}(f_h),\quad J_h=J_{A_h}(f_h),\quad
I_{\rm rem}=\sum_{h\text{ internal}}\pi_h I_{A_h}(f_h),\quad
I_{\rm term}=\sum_{l\text{ leaf}}\pi_l I(f_l).
\]
\begin{theorem}[C019 telescope and conditional endpoint]
\label{thm:tree}
For every such tree,
\[
H(f)=\sum_h\pi_h(B_h+J_h)+\sum_l\pi_lH(f_l),\quad
I(f)=I_{\rm rem}+I_{\rm term},\quad
\sum_h\pi_hJ_h\le c_0 I_{\rm rem}.
\]
If $K,C_{\rm leaf}\ge0$ satisfy
$\sum_h\pi_hB_h\le K I_{\rm rem}$ and
$H(f_l)\le C_{\rm leaf} I(f_l)$ for every positive-reach leaf,
then $H(f)\le\max(K+c_0,C_{\rm leaf})I(f)$.
\end{theorem}
\begin{proof}
Multiply each identity of Lemma~\ref{lem:block} by $\pi_h$.
Since $\pi_{hx}=\pi_h\alpha_h$, all interior child terms cancel
on the finite tree; only the root and terminal terms remain.
Summing its channel bound proves the inequality. Substitute the
two payment hypotheses and influence conservation. These are actual
reach weights, not unweighted counts of nodes. Influence removed at
ancestors is absent from the residual child budget and cannot be
spent again.
\end{proof}

\begin{lemma}[B06 R2: leaf-partition invariance]
\label{lem:r2}
Write $\Pi_H=H(f)-\E_lH(f_l)$ and
$\Pi_I=I(f)-\E_lI(f_l)$. For every real $K$,
\[
KI(f)-H(f)=\E_l[KI(f_l)-H(f_l)]+K\Pi_I-\Pi_H.
\]
Moreover $\Pi_I$ equals the expected number of queried coordinates
that are pivotal for the original $f$ at the input on its path.
The aggregate entropy drop and influence removed depend only on
the actual physical leaf partition, not on the query order that
realizes that partition. For constant leaves,
$\sum_h\pi_hB_h\ge H(f)-c_0I(f)$.
\end{lemma}
\begin{proof}
The deficit identity is subtraction of the two telescopes in
Theorem~\ref{thm:tree}. At any node, flipping a free coordinate
preserves its ancestor assignment, so its actual restricted pivotal
indicator is the original pivotal indicator conditioned on that
node. Multiplying by reach probability and summing each queried
coordinate once proves the path interpretation, including block
queries. For a fixed terminal assignment subcube, the residual
function and its entropy/influence do not depend on the order of
queries. Finally use $\Pi_H=\sum\pi(B+J)$ and
$\sum\pi J\le c_0\Pi_I$, with zero terminal entropy/influence.
\end{proof}
This is B06-LEAD-R2-LEAF-PARTITION-INVARIANCE version 1,
with the block extension derived from the same identities.
The individual erasure/channel/posterior-credit totals need not be
invariant: only their full entropy-drop and influence combination is.

\section{A097: complete references and their compatible telescope}

For a probability $P$ and a supported subprobability $Q$, define
\[
\CE(P,Q)=\sum P\log_2(1/Q),\qquad
\KL(P\|Q)=\sum P\log_2(P/Q)=\CE(P,Q)-H(P).
\]
Support means $Q(s)>0$ whenever $P(s)>0$. Gibbs' inequality holds
for this generalized divergence: complete $Q$ by a failure atom
and assign that atom zero $P$ mass.

\begin{lemma}[Complete nonempty reference interface]
\label{lem:complete}
Let $v=P(S\ne\varnothing)$ and let $R$ be a subprobability on
nonempty sets positive on their actual support. Put
$Q(\varnothing)=1-v$ and $Q(S)=vR(S)$ for nonempty $S$.
Then $Q$ is complete and
\[
\CE(P,Q)=h_2(v)+\sum_{S\ne\varnothing}P(S)\log_2(1/R(S)).
\]
For $v=0$, choose $Q=\delta_\varnothing$; for $v=1$ the empty
mass is zero.
\end{lemma}
\begin{proof}
The total mass is at most $1-v+v=1$. Expand the cross-entropy and
collect $-(1-v)\log_2(1-v)-v\log_2v=h_2(v)$.
\end{proof}

At each leaf of a fixed physical tree choose any complete reference
$Q_l$ supported on $P_l=P_{f_l}$. Its total mass is at most one;
it may depend on the terminal history but is fixed before sampling
a Fourier set. Define references recursively upwards as follows.
At a node with block $A$, complement $C$, and $m=2^{|A|}$, write
\[
q(T)=\sum_{U\subseteq A}a_{U\cup T}^2=m^{-1}\sum_xP_x(T),\quad
r(T)=m^{-1}\sum_xQ_x(T),\quad
\theta(U\mid T)=a_{U\cup T}^2/q(T),
\]
\[
Q_h(U\cup T)=\theta(U\mid T)r(T).\tag{G}\label{eq:G}
\]
When $q(T)=0$ fix any normalized $\theta$, for example a point
mass at $U=\varnothing$. Reference mass on such a fiber must be
kept, even though its true mass vanishes. At $q(T)>0$ put
\[
k_P(x\mid T)=P_x(T)/(mq(T)),\qquad
k_Q(x\mid T)=Q_x(T)/(mr(T)).
\]
\[
\kappa_h=\sum_Tq(T)\KL(k_P(\cdot\mid T)\|k_Q(\cdot\mid T)).
\]
These are normalized posteriors on each positive true fiber.

\begin{theorem}[Compatible reference transfer, A097 version 1]
\label{thm:reference}
The recursive $Q_h$ are supported complete subprobabilities and
\[
\CE(P_f,Q_{\rm root})
=\sum_h\pi_h(B_h+J_h-\kappa_h)+\sum_l\pi_l\CE(P_l,Q_l),\tag{T}
\]
\[
\KL(P_f\|Q_{\rm root})
=\sum_l\pi_l\KL(P_l\|Q_l)-\sum_h\pi_h\kappa_h.\tag{S}
\]
Consequently $0\le\sum_h\pi_h\kappa_h\le
\sum_l\pi_l\KL(P_l\|Q_l)$.
If $\CE(P_l,Q_l)\le C_{\rm leaf}I(f_l)$ and
\[
\sum_h\pi_h(B_h-\kappa_h)\le K I_{\rm rem},\tag{R}\label{eq:R}
\]
then $H(f)\le\max(K+c_0,C_{\rm leaf})I(f)$.
\end{theorem}
\begin{proof}
From (G), $\sum Q_h=m^{-1}\sum_x\sum Q_x\le1$.
If $a_{U\cup T}^2>0$, some $P_x(T)>0$; inductive support gives
$r(T)>0$ and $Q_h(U\cup T)>0$.
For the true/reference laws on $(x,T)$ with masses $P_x(T)/m$
and $Q_x(T)/m$, expansion first by $x$, then by $T$, gives
\[
m^{-1}\sum_x\KL(P_x\|Q_x)=\KL(q\|r)+\kappa_h.
\]
The true parent and reference share $\theta$, whence
$\KL(P_h\|Q_h)=\KL(q\|r)$.
Combine this with the block entropy identity to get
\[
\CE(P_h,Q_h)=B_h+J_h+m^{-1}\sum_x\CE(P_x,Q_x)-\kappa_h.
\]
Finite telescoping proves (T). Telescope the divergence identity
to obtain (S); nonnegativity proves the stated credit ceiling.
Finally use $H(f)\le\CE(P_f,Q_{\rm root})$, the paid channel
bound, (R), and influence conservation.
\end{proof}

\begin{conjecture}[OPEN-REF-TRANSFER: universal complete-reference payment]
\label{conj:reference}
There exist finite $K,C_{\rm leaf}\ge0$ such that for every
$n\ge0$ and every Boolean $f$ there exist a finite actual stopping
tree and supported complete leaf subprobabilities $Q_l$ satisfying
$\CE(P_l,Q_l)\le C_{\rm leaf}I(f_l)$ for every leaf and (R),

\end{conjecture}
Neither Gibbs positivity nor the existence of $Q_l$ proves (R).
For $Q_l=P_l$, all interior references equal the actual spectra,
all $\kappa_h$ vanish, and (T) is exactly the entropy telescope.
Positive reference credit is constrained by terminal divergence
slack; it is not an additional freely reusable physical influence.
An interior reference reset is outside (G) and would require its
own correction. C020 costs, if inserted at leaves via
Lemma~\ref{lem:complete}, retain their empty-atom entropy, root
normalizer, scale prior, labels, ranks, attachments and all other
complete-description terms.

\section{Closed B05 compatible-reference calibration, A097 version 1}
\label{sec:b05ref}
Let even $k\ge2$, $\delta=2^{-k}$, $\epsilon=2^{-k/2}$,
$r=2k$, $p=2^{-r}$, and let $b$ be the quadratic bent sign
on $k$ data bits. Set $b'=b-2\mathbf1_0$, let $h$ be their fair
physical-selector splice, and put $F=1+A_r(h-1)$.
The tree queries controls in order, stops at the first inactive
assignment, otherwise queries the selector and stops at $b$ or $b'$.
Constant leaves use $\delta_\varnothing$; seed references are either
their true laws (mode E) or the same uniform subset law $Q(T)=\delta$
(mode U). These are complete references fixed before sampling.
\begin{proposition}[A097-B05-COMPATIBLE-REFERENCE-CERTIFICATE version 1]
\label{prop:b05ref}
On this precise tree, for every even $k\ge2$,
\[
 B_{\rm tree}:=\sum_h\pi_hB_h\le3 I_{\rm rem},\quad
 I_{\rm rem}=p(rc+\delta),\quad I_{\rm term}=p(k/2+k\delta),
 \quad c=1-\epsilon+\delta.
\]
Mode E has zero posterior credit. Mode U has
\[
0\le\frac{K_{\rm tree}}{I_{\rm rem}}
\le\frac{8\delta(1-\delta)}{(rc+\delta)\ln2},
\quad K_{\rm tree}=\sum_h\pi_h\kappa_h.
\]
Both erasure and erasure-minus-credit rates tend to 2, and
the complete root reference satisfies
\[
\CE(P_F,Q_F)\le(3+c_0)I_{\rm rem}+2I_{\rm term}
\le(3+c_0)I(F).
\]
\end{proposition}
\begin{proof}
The bent coefficients are $\sigma\epsilon$, with counts
$N_\pm=(2^k\pm2^{k/2})/2$, and the point flip subtracts $2\delta$
from each. The selector has coefficients $\sigma\epsilon-\delta$
and $\delta$; hence
\[
q_\sigma=\delta-2\sigma\delta\epsilon+2\delta^2,\quad
B_{\rm sel}=\sum_\sigma N_\sigma
                    \phi((\sigma\epsilon-\delta)^2,\delta^2).
\]
The mean of $h$ is $\mu=\epsilon-\delta$; $c=1-\mu\in[3/4,1)$.
At the active suffix with $s$ controls, let $u_s=2^{-s}$ and
$d_s=u_sc$. The physical reach probability is $p/u_s$.
The control's residual-empty squared masses are
$(1-d_s)^2,d_s^2$, and every other packet has fair spectral
membership. Thus with $q_s=1-2d_s+2d_s^2$,
\[
B_s=\phi((1-d_s)^2,d_s^2)+(1-q_s),\quad
B_{\rm tree}=p\left[B_{\rm sel}+\sum_{s=1}^r B_s/u_s\right].
\]
The control influences are $pc$, the selector influence is
$p\delta$, and the seed influences are $k/2$ and $k/2+2k\delta$
because each neighbor of the flipped point has value $+1$.
This proves both influence formulas, with terminal budget retained.

For clarity the actual compatible reference correction is explicit.
If $z_{s-1}$ is the active-child reference vacuum mass, (G) gives
\[
z_s=\frac{(1-d_s)^2}{q_s}\frac{1+z_{s-1}}2,\quad
\kappa_s=q_s\KL\left(
 \frac{(1,(1-2d_s)^2)}{2q_s}\,
 \middle\|\,\frac{(1,z_{s-1})}{1+z_{s-1}}\right).
\]
All nonempty packets have both physical posteriors concentrated on
the active child, hence zero correction.
For mode E, $z_0=\mu^2$ and every correction vanishes.
For mode U, $z_0=\delta\mu^2/q_+$ and
$\kappa_{\rm sel}=J_{\rm sel}$, where
$J_{\rm sel}=H(q)-[k+H(b')]/2$.
The exact requested rate is therefore
\[
\frac{B_{\rm sel}-\kappa_{\rm sel}
       +\sum_{s=1}^{2k}(B_s-\kappa_s)/u_s}{2kc+\delta}.
\]

For $d_s\le1/2$, the root-packet bound gives
\[
B_s/u_s\le2c+u_sc^2[\log_2e+2s-2\log_2c].
\]
Use $\sum u_s\le1$, $\sum su_s\le2$, and
$B_{\rm sel}\le h_2(\delta)\le\delta(k+\log_2e)$ to obtain
\[
B_{\rm tree}/p\le2rc+C_0+\delta(k+\log_2e),\quad
C_0=\log_2e+4+2\log_2(4/3)<13/2.
\]
The last comparison follows, for example, from
$\ln2>69/100$ and $\ln(4/3)<3/10$:
the latter is certified by the first five alternating terms of
$\ln(1+1/3)$, an upper bound less than $3/10$.
For even $k\ge6$, $rc\ge9$ and
$\delta(k+3/2)\le15/128$, giving
$B_{\rm tree}/I_{\rm rem}<2+(13/2+15/128)/9<3$.
The two base cases have the rational certified upper bounds
\[
\begin{split}
 k=2:\quad&B_{\rm tree}/I_{\rm rem}
 \le\frac{1529024733915133429860119}{604462909807314587353088}<\frac{253}{100},\\
 k=4:\quad&B_{\rm tree}/I_{\rm rem}
 \le\frac{1419432708511339676246917}{604462909807314587353088}<\frac{235}{100}.
\end{split}
\]
These are the exact A097 output intervals from its rational
32-term atanh enclosure; provenance and copied endpoints are in
\path{out/B05_CERTIFICATE_INTAKE.json}.
This explicitly isolates the finite certificate dependency from
the analytic all-$k$ tail. The lower bound
$B_s\ge2u_sc-2u_s^2c^2$ gives
$B_{\rm tree}/p\ge2rc-2$, proving the rate limit.

For mode U, the flipped bent spectrum has
$\chi^2(P_{b'}\|U)=16\delta(1-\delta)$:
its atom difference is $\delta(-4\sigma\epsilon+4\delta)$,
and the uniform mean of $\sigma$ is $\epsilon$.
The inequality $\ln t\le t-1$ gives
$\KL(P_{b'}\|U)\le\chi^2/\ln2$.
The terminal slack is $p\KL(P_{b'}\|U)/2$, so (S) proves
the credit cap and its negligible rate. Seed reference CE is at
most $k\le2I(\text{seed})$ in both modes: for E, entropy is at
most $k$; for U, cross-entropy equals $k$.
Theorem~\ref{thm:reference} completes the root CE bound.
\end{proof}

\section{B05: scope-correct Boolean local obstruction}\label{sec:b05}
\begin{proposition}[B05 version 1; B19 accepted at stated scope]
\label{prop:b05}
For every finite $K,A_0,B_0\ge0$ there is a genuine Boolean
singleton split with children $G,H$, and
\[
q_W=(\widehat G(W)^2+\widehat H(W)^2)/2,\quad
e_W=(\widehat G(W)-\widehat H(W))^2/4,\quad
\vartheta_W=e_W/q_W,\quad d=\sum_We_W,
\]
such that
\[
B_y:=\sum_Wq_Wh_2(\vartheta_W)>Kd+A_0V+B_0M,\quad
V=\sum_Wq_W(\vartheta_W-d)^2,\quad
M=\sum_j(d_j(G)-d_j(H))^2.
\]
Thus a fixed-price bound of this form cannot hold at every coordinate.
\end{proposition}
\begin{proof}
Use $0/1$ physical labels for this construction; this relabeling
does not change influence or squared Fourier coefficients.
Let even $k\ge4$, $\delta=2^{-k}$, $p=2^{-2k}$,
$d=p\delta=2^{-3k}$,
\[
b(x)=(-1)^{x_1x_2+\cdots+x_{k-1}x_k},\quad
b'(x)=b(x)-2\mathbf1_{x=0},\quad A(z)=\mathbf1_{z=0},
\]
on $k$ data bits and $2k$ controls, and set
$G=1+A(b-1)$, $H=1+A(b'-1)$.
These are Boolean, since $b(0)=1$ and the point flip exchanges
that value with $-1$. They disagree at one point of a $3k$ cube.
Every Walsh coefficient $c_T$ of $b$ has magnitude $\sqrt\delta$:
each quadratic pair has four coefficients of magnitude $1/2$,
and the pairs tensorize. Also $c_\varnothing=\sqrt\delta$.
The point flip and activation give a coefficient difference $2d$
for all $2^{3k}$ residual packets, so $e_W=d^2$ and $\sum e=d$.
For nonempty data index $T$ the coefficient midpoint is
$p(c_T-\delta)$, and
\[
q_W/e_W=(c_T/\delta-1)^2+1\ge1/(4\delta).
\]
For empty $T$ and nonempty control index the midpoint is
$p(\sqrt\delta-1-\delta)$, giving the same bound.
For the all-empty index it is $1+p(\sqrt\delta-1-\delta)$,
whose magnitude is at least $1-p$; its ratio is larger still.
Hence $\max\vartheta\le4\delta$.
Since $h_2(t)\ge t\log_2(1/t)$, summing just the
$(2^k-1)2^{2k}$ nonempty-data packets gives
$B_y\ge(1-\delta)d(k-2)$.
Furthermore $V=\sum q\vartheta^2-d^2\le4\delta d$.
Every one-bit neighbor of the flipped point has common value $+1$.
Exactly one undirected edge per residual coordinate changes pivotal
status; the directed-edge convention gives $d_j(H)-d_j(G)=2d$.
Thus $M=3k(2d)^2=12kd^2$.
Consequently
\[
(B_y-Kd-A_0V-B_0M)/d
\ge(1-\delta)(k-2)-K-4A_0\delta-12B_0kd\longrightarrow\infty.
\]
Finally the parent's queried-coordinate spectral probability in
fiber $W$ is $e_W/q_W$, so this is precisely its $B_y$.
\end{proof}
This does not refute Conjectures~\ref{conj:universal},
\ref{conj:tree}, or \ref{conj:reference}. Its quantifier is an
adversarial \emph{specified} coordinate; the reduction chooses a
coordinate or a whole tree. Child influence is of order $kp$,
whereas the bad local charge is of order $kp\delta$.
No unbounded $H/I$ Boolean family follows.
