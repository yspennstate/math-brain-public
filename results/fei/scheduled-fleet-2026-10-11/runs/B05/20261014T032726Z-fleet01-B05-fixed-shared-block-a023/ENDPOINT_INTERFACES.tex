\qquad
\Lambda_X(f):=|X|q(E_X^f(\varnothing))
 +\sum_{i\notin X}q(E_X^f(\{i\})).
\]
For Boolean $f(X,U)$ and $g(X,V)$ with $U\cap V=\varnothing$,
\[
\Phi(fg)\le
\sum_{i\in U}q(E_X^f(i))+\sum_{j\in V}q(E_X^g(j))
+|X|q\!\left(\min\{E_X^f(\varnothing),E_X^g(\varnothing)\}\right)
\le \Lambda_X(f)+\Lambda_X(g).
\]

\[
\mathbb E_{\sigma}\Lambda_{X\setminus\{x\}}(f\mid x=\sigma)
\le\Lambda_X(f),
\qquad
\mathbb E_{\xi\in\{-1,1\}^X}\Phi(f\mid X=\xi)
\le\Lambda_X(f).
\]
