\section{Conventions and mathematical scope}
For every finite $n\ge0$, $f:\{-1,1\}^n\to\{-1,1\}$ is Boolean,
$a_S=\E f\chi_S$, $P_f(S)=p_S=a_S^2$,
$H(f)=\sum_Sp_S\log_2(1/p_S)$, and
$I(f)=\sum_S|S|p_S=\sum_i d_i(f)$, where
$d_i(f)=\Pr(f(x)\ne f(x^{\oplus i}))$.
Every unmarked entropy, cross-entropy and divergence is in bits;
$c_0=2/\ln2$. Zero entropy atoms contribute zero. Spectral conditioning
does not mean physical restriction. We write
\[
 B_i(f)=H(S_i\mid S_{-i}),\quad B(f)=\sum_i B_i(f),\quad
 \mathcal L(f)=\sum_{S:p_S>0}p_S\sum_{i\in S}
                    \log_2(1+p_{S\setminus\{i\}}/p_S).
\]
The symbol $\mathcal L$ is the aggregate C018 downward charge. B05's
quantity named $L$ is instead a single $B_i$; Section~\ref{sec:b05}
calls it $B_y$ to prevent an erroneous substitution.
\[
C_{\rm FEI}=\sup_{n,f:I(f)>0} H(f)/I(f),\quad
K_E=\sup_{n,f:I(f)>0} B(f)/I(f),\quad
K_{\mathcal L}=\sup_{n,f:I(f)>0}\mathcal L(f)/I(f).
\]
These suprema may be infinite and need not be attained.

\section{C005: actual singleton restrictions}
\begin{lemma}[Entropy and influence identities, source ER Sections 1--4]
\label{lem:single}
For every Boolean $f$ and coordinate $i$, write $f_\pm=f_{i=\pm1}$.
Then
\[
H(f)-\tfrac12(H(f_+)+H(f_-))=B_i(f)+J_i(f),\qquad
I(f)=d_i(f)+\tfrac12(I(f_+)+I(f_-)),
\]
where $J_i$ is the Jensen--Shannon divergence of $P_{f_+},P_{f_-}$,
and
\[
0\le J_i\le c_0\sum_{T\not\ni i}\min(p_T,p_{T\cup\{i\}})
             \le c_0\min(d_i,1-d_i).
\]
\end{lemma}
\begin{proof}
Set $u=a_T$, $v=a_{T\cup\{i\}}$, $m=u^2+v^2$.
The two actual coefficients are $u+v,u-v$ and their mean squared
mass is $m$. The chain rule gives
$B_i=\sum_T[h(u^2)+h(v^2)-h(m)]$, where $h(t)=-t\log_2t$.
Meanwhile
$J_i=\sum_T[h(m)-\tfrac12h((u+v)^2)-\tfrac12h((u-v)^2)]$.
Adding cancels $h(m)$ and proves the identity. The child laws are
normalized by Booleanity and Parseval, so $J_i\ge0$ by concavity.

For $0\le z\le1$, let
$D(z)=\tfrac12[(1+z)\ln(1+z)+(1-z)\ln(1-z)]$.
Then $D(z)\le1-\sqrt{1-z^2}$: both vanish at zero;
their derivatives are $\operatorname{atanh}z$ and
$z/\sqrt{1-z^2}$, and the derivative of the difference of these
derivatives is $(1-z^2)^{-3/2}-(1-z^2)^{-1}\ge0$.
Continuity handles $z=1$.
At $m>0$ the JS contribution is $mD(2|uv|/m)/\ln2$,
at most $(m-|u^2-v^2|)/\ln2=c_0\min(u^2,v^2)$.
Zero fibers vanish. Sum over $T$ and use
$\sum_Tv^2=d_i$, $\sum_Tu^2=1-d_i$.
For $j\ne i$, averaging the $j$-pivotal indicator over the fixed
$i$ coordinate gives $d_j(f)=\tfrac12(d_j(f_+)+d_j(f_-))$.
Summation proves the influence identity.
\end{proof}

\section{C006: actual blocks and their channel price}
\begin{lemma}[Cube log-Sobolev inequality in the required normalization]
\label{lem:lsi}
For $g\ge0$ on a finite uniform cube,
\[
 \Ent(g^2):=\E[g^2\ln(g^2/\E g^2)]
 \le\tfrac12\sum_i\E(g(x)-g(x^{\oplus i}))^2.
\]
\end{lemma}
\begin{proof}
On one bit, by homogeneity the required inequality for values $t,1$
is $F(t)\ge0$, where
\[
F(t)=(t^2+1)\ln((t^2+1)/2)-2t^2\ln t+(t-1)^2.
\]
Here $F(1)=0$ and
$F'(t)=2t\ln((t^2+1)/(2t^2))+2(t-1)$.
For $t\ge1$, the difference
$1-1/t-\ln(2t^2/(t^2+1))$ is zero at 1 and has derivative
$(t-1)^2/[t^2(t^2+1)]\ge0$, proving $F'\ge0$.
The identity $F(t)=t^2F(1/t)$ handles $t<1$, and continuity
handles zero values.

For completeness, entropy tensorization follows as follows. For a
nonnegative $w$ with mean $m>0$, put $R(x)=w(x)2^{-n}/m$.
Then $\Ent(w)/m=D_{\rm nat}(R\|U)$.
The divergence chain rule writes this as the sum of the divergences
of $R(X_i\mid X_{<i})$ from the uniform one-bit law.
Replacing $X_{<i}$ by $X_{-i}$ increases each mean divergence,
because conditioning decreases entropy. Direct substitution of the
density shows that the resulting $i$th term, multiplied by $m$, is
$\E_{U,-i}\Ent_{U,i}(w)$.
Thus $\Ent(w)\le\sum_i\E_{-i}\Ent_i(w)$.
Apply the one-bit bound to $w=g^2$ on each fiber. The case $m=0$
is immediate.
\end{proof}

\begin{lemma}[Block identity and paid channel, source BR Sections 1--3]
\label{lem:block}
Let $A$ be a nonempty physical block, $C=[n]\setminus A$,
$X$ a uniform assignment on $A$, and conditionally on $X=x$ let
$T$ have the spectral law of $f_x$. Then
\[
H(f)-\E_xH(f_x)=B_A(f)+J_A(f),\quad
B_A(f)=H(S_A\mid S_C),\quad J_A(f)=I(X;T),
\]
\[
0\le J_A(f)\le c_0 I_A(f),\qquad
I(f)=I_A(f)+\E_xI(f_x),\quad I_A=\sum_{i\in A}d_i.
\]
\end{lemma}
\begin{proof}
Let $v_T(x)=\E_y f(x,y)\chi_T(y)$,
$P_x(T)=v_T(x)^2$, and $q(T)=\E_xP_x(T)$.
Parseval in $x$ gives $q(T)=\sum_{U\subseteq A}a_{U\cup T}^2$,
the actual $C$-marginal. Hence
$H(f)=H(q)+H(S_A\mid S_C)$ and
$J_A=H(q)-\E_xH(P_x)$, proving the first identity.
For $q(T)>0$, apply Lemma~\ref{lem:lsi} to
$g_T(x)=|v_T(x)|/\sqrt{q(T)}$, multiply by $q(T)$ and sum.
The left side is $(\ln2)J_A$; the right side is at most
\[
\tfrac12\sum_{i\in A}\E_x\sum_T(v_T(x)-v_T(x^{\oplus i}))^2
=\tfrac12\sum_{i\in A}\E_{x,y}(f(x,y)-f(x^{\oplus i},y))^2
=2I_A(f).
\]
The middle equality is Parseval on $C$; the last uses Boolean
values. Zero $q$ fibers vanish. Averaging each pivotal indicator
on $C$ over $X$ proves influence conservation.
\end{proof}

\begin{lemma}[B06 R1: block-to-singleton, one canonical source ID]
\label{lem:r1}
For any law on subsets and any block $A$,
\[
H(S_A\mid S_{A^c})\ge\sum_{i\in A}H(S_i\mid S_{-i}).
\]
For Boolean $f$ with $I_A(f)>0$, a bound $B_A(f)\le KI_A(f)$
therefore gives some relevant $i\in A$ with $B_i(f)\le Kd_i(f)$.
A bound $\Delta_AH(f)\le KI_A(f)$ gives some such $i$ with
$\Delta_iH(f)\le(K+c_0)d_i(f)$.
\end{lemma}
\begin{proof}
Order $A$ and expand the block conditional entropy by the chain
rule. Each term conditions only on $A^c$ and preceding bits of
$A$; conditioning further on all other bits can only reduce it.
Sum, then apply influence-weighted averaging. For the final
statement $B_A\le\Delta_AH$ by Lemma~\ref{lem:block}, and
$\Delta_iH\le B_i+c_0d_i$ by Lemma~\ref{lem:single}.
Irrelevant coordinates have zero $B_i$.
\end{proof}
This is B06-LEAD-R1-BLOCK-TO-SINGLETON version 1, reproduced here.
A block cannot rescue the same node if every singleton erasure
rate fails; its distinct role is skipping intermediate restrictions.

\section{C018: the proved equivalence and its open Boolean leaf}
\begin{lemma}[Erasure comparison and downward charge, ER Sections 3--6]
\label{lem:downward}
For every Boolean $f$,
\[
\mathcal L(f)\le B(f)\le\mathcal L(f)+I(f)/\ln2,\qquad B(f)\le H(f).
\]
The coefficients also satisfy
\[
\sum_Sa_S^2=1,\qquad \sum_Sa_Sa_{S\triangle U}=0
\quad(\varnothing\ne U\subseteq[n]).
\]
Conversely these identities for real $a$ imply its inverse Walsh
transform is Boolean.
\end{lemma}
\begin{proof}
For nonnegative $r,s$,
$h(r)+h(s)-h(r+s)=r\log_2(1+s/r)+s\log_2(1+r/s)$,
with continuous endpoint conventions. On each edge $(T,T\cup\{i\})$
the second term is the downward charge. The first is at most
$s/\ln2$, by $\ln(1+t)\le t$, including zero endpoints by continuity.
Sum to prove the sandwich. In any coordinate order,
$H(S_i\mid S_{-i})\le H(S_i\mid S_{<i})$; the entropy chain
rule proves $B\le H$. Fourier transforming $f^2=1$ gives the
convolution identities. Conversely they say all nonconstant
coefficients of the square of the inverse transform vanish and its
constant coefficient is 1, so its square is pointwise 1.
\end{proof}

\begin{theorem}[C018; implication proved, payment open]
\label{thm:erasure}
In the extended real sense,
\[
K_E\le C_{\rm FEI}\le K_E+c_0,\qquad
K_{\mathcal L}\le K_E\le K_{\mathcal L}+1/\ln2.
\]
In particular any uniform bound $B(f)\le KI(f)$ implies
$H(f)\le(K+c_0)I(f)$; any uniform bound
$\mathcal L(f)\le KI(f)$ implies
$H(f)\le(K+3/\ln2)I(f)$.
\end{theorem}
\begin{proof}
The lower bound follows from Lemma~\ref{lem:downward}.
Assume $K_E<\infty$. Summing Lemma~\ref{lem:single} gives
$\sum_i\Delta_iH\le(K_E+c_0)I(f)$.
Irrelevant coordinates have $B_i=J_i=\Delta_iH=0$.
If $f$ is nonconstant, some relevant coordinate satisfies
$\Delta_iH\le(K_E+c_0)d_i$ by weighted averaging.
Induct on dimension, restricting that coordinate, and use the exact
influence identity to obtain $H(f)\le(K_E+c_0)I(f)$.
Constant functions have $H=I=0$. If $K_E=\infty$ the upper
comparison is vacuous. The remaining comparisons and constant follow
from the sandwich.
\end{proof}

\section{Paid universal layers and the exact remaining erasure}
Put $\phi(r,s)=h(r)+h(s)-h(r+s)$ and define
\[
B_0=\sum_i\phi(a_\varnothing^2,a_{\{i\}}^2),\quad
T_{12}=\sum_i\sum_{j\ne i}\phi(a_{\{i\}}^2,a_{\{i,j\}}^2),\quad
R_{\ge2}=\sum_{|S|\ge2}\sum_{i\notin S}\phi(a_S^2,a_{S\cup\{i\}}^2).
\]
Thus $B=B_0+T_{12}+R_{\ge2}$ exactly, with each cube edge
counted once from its lower set.
\begin{lemma}[B20 root-pair payment, version 1]
\label{lem:root}
For every Boolean $f$, $B_0\le c_r I(f)$, where
$c_r=2/(\sqrt e\ln2)$.
\end{lemma}
\begin{proof}
For $r+s\le1$, $\phi(r,s)\le s\log_2(e/s)$, by
$r\ln(1+s/r)\le s$ and $\ln((r+s)/s)\le\ln(1/s)$.
The physical derivative identity gives $|a_{\{i\}}|\le d_i(f)$.
The function $s\log_2(e/s)$ is increasing on $(0,1]$, so its
value at $a_{\{i\}}^2$ is at most $d_i^2\log_2(e/d_i^2)$.
Finally $\sup_{0<t\le1}t\log_2(e/t^2)=2/(\sqrt e\ln2)$,
as differentiation shows; sum over $i$.
\end{proof}

\begin{theorem}[A096 universal layer payment, version 1]
\label{thm:level12}
For every Boolean $f$, writing
$W_2=\sum_{i<j}a_{\{i,j\}}^2$ and
$H_2=\sum_{i<j}a_{\{i,j\}}^2\log_2(1/a_{\{i,j\}}^2)$,
\[
H_2\le\frac{155600}{50301}I(f),\qquad
T_{12}\le c_{12}I(f),\quad c_{12}=\frac{16700}{2187}<8.
\]
\end{theorem}
\begin{proof}
Let $\epsilon_i=d_i(f)$ and
$h_i=(f_{i=+1}-f_{i=-1})/2$. Booleanity implies
$h_i\in\{0,\pm1\}$ and $\Pr(h_i\ne0)=\E h_i^2=\epsilon_i$.
For $L_i=\sum_{j\ne i}a_{\{i,j\}}x_j$, let
$u_i=\E L_i^2=\E h_iL_i=\sum_{j\ne i}a_{\{i,j\}}^2$.
Independence and $\cosh v\le e^{v^2/2}$ give
$\E e^{t|L_i|}\le2e^{t^2u_i/2}$.
The scalar bound follows by integrating $\tanh v\le v$.
Conditioning on $h_i\ne0$ and applying Jensen gives
\[
u_i\le\epsilon_i\left(\frac{\ln(2/\epsilon_i)}t+\frac{tu_i}2\right),
\quad\text{hence}\quad
u_i\le2\epsilon_i^2\ln(2/\epsilon_i)
\]
after optimization; zero support or row energy is immediate.

For $I,W_2>0$, the unordered-pair reference
$Q(\{i,j\})=2\epsilon_i\epsilon_j/I^2$ has total mass at most
one and is positive on all occupied pairs. Gibbs for the normalized
level law $a_{\{i,j\}}^2/W_2$, retaining its level mass, gives
\[
H_2\le\sum_i u_i\log_2(1/\epsilon_i)
                 +W_2\log_2(I^2/(2W_2)).
\]
Using $t=-\ln\epsilon_i$ and the maxima
$\sup t^2e^{-t}=4/e^2$, $\sup te^{-t}=1/e$, the first sum is
at most $[8/(e^2\ln2)+2/e]I$.
For every $z>0$, $\ln z\le2\sqrt z/e$, so the second term
is at most $\sqrt2 I\sqrt{W_2}/(e\ln2)\le\sqrt2 I/(e\ln2)$.
This includes negative logarithms. Thus $H_2\le K_2I$ for
$K_2=8/(e^2\ln2)+2/e+\sqrt2/(e\ln2)$.
Since every edge has $r+s\le1$, its cost is at most
$s\log_2(1/s)+s/\ln2$. Each pair is counted twice and
$2W_2\le I$, giving $T_{12}\le(2K_2+1/\ln2)I$.
The rational comparisons $e>27/10$, $\ln2>69/100$ and
$\sqrt2<71/50$ follow respectively from the exponential series
through degree five, the first two positive atanh terms at $1/3$,
and squaring. Substitution yields
$K_2<155600/50301$ and $2K_2+1/\ln2<16700/2187$.
Constants or $W_2=0$ give zero contributions.
\end{proof}
The canonical ID is A096-UNIVERSAL-LEVEL12-ERASURE version 1.
A098 independently accepted precisely this all-dimensional layer;
that review does not pay the sum of all higher levels.

\section{B20 low-total-influence completeness, strengthened version 2}
\label{sec:low}
For finite-valued $s:\mathbb N\to[0,\infty)$ with $s(N)=o(N)$,
let $K_E(s)$ be the supremum $B(F)/I(F)$ over every dimension
$N$ and every Boolean $F$ with $0<I(F)\le2^{-s(N)}$.
\begin{theorem}[B20-ERASURE-LOW-INFLUENCE-COMPLETE, proposed version 2]
\label{thm:low}
In the extended reals, $K_E(s)=K_E$. Consequently
$C_{\rm FEI}\le K_E(s)+2/\ln2$.
\end{theorem}
\begin{proof}
Disjoint products have product Fourier laws; conditional entropies
and physical influence add. For a nonconstant seed $f$ on $n\ge1$
bits take $m$ copies and a fresh fair sign $z$, and put
$h_m=z\prod_{j=1}^m f(x^{(j)})$.
Then $\E h_m=0$, $B(h_m)=mB(f)$ and $I(h_m)=mI(f)+1$.
The $z$-frequency is deterministic and its erasure is zero.

Activate $h_m$ at one point of $k\ge1$ controls: $F=1+A_k(h_m-1)$,
$\delta=2^{-k}$. The constant coefficient is $1-\delta$;
every nonempty core coefficient is $\delta\widehat h_m(S)$
in every control-frequency slice; empty-core/nonempty-control
coefficients have magnitude $\delta$. Partition the core-coordinate
edges by their residual core index. With $b_i=\widehat h_m(\{i\})^2$,
the original root packet is $\phi(0,b_i)=0$ and
\[
B_{\rm data}(F)=\delta B(h_m)
 +\delta(1-\delta)\sum_i\phi(1,b_i)
 +\sum_i\phi((1-\delta)^2,\delta^2b_i)\ge\delta mB(f).
\]
Control erasures are nonnegative. Direct physical edge counts give
$I(F)=\delta[mI(f)+1+k]$, since $h_m$ is balanced.

Set $\bar s(N)=\max_{M\le N}s(M)$; it is nondecreasing and
$o(N)$, because the finite initial maximum is $o(N)$ and the
tail is bounded by $\varepsilon N$ for every fixed $\varepsilon>0$.
Define $k_m=\lceil\bar s(2mn)+\log_2(3mn)\rceil+1$ and
$N_m=mn+1+k_m$. For large $m$, $k_m\le mn-1$, $N_m\le2mn$, and
\[
0<I(F_m)\le2^{-k_m}3mn
\le2^{-\bar s(2mn)}\le2^{-\bar s(N_m)}.
\]
If $K=K_E(\bar s)$ is finite, admissibility implies
$mB(f)\le K[mI(f)+1+k_m]$.
Divide by $m$ and let $m\to\infty$; $k_m/m\to0$ gives
$B(f)\le KI(f)$. Hence
$K_E\le K_E(\bar s)\le K_E(s)\le K_E$.
If $K_E=\infty$, the same argument excludes finite $K_E(s)$;
the extended-real equality follows. C018 proves the corollary.
\end{proof}
This refines the source's valid additive bound
$K_E\le K_E(s)+c_r$; the original version is not refuted.
The fresh balancing makes the root-pair correction zero, while its
one unit of influence disappears after tensor amplification.
The proposed version has a complete proof and exact finite identity
validation and an independent A090 scoped acceptance; the A094 second
review remains pending.
This is aggregate erasure transport. It makes no same-constant
claim about a \emph{selected} data-coordinate witness; A005's
control-witness counterexample therefore remains applicable.
Finiteness of the low-influence constant is still unproved.

\section{A090: all fixed higher tails retain the optimal erasure constant}
\label{sec:tail}
For a fixed integer $\ell\ge2$ let $R_{\ge\ell}$ retain the edge costs
$\phi(p_T,p_{T+i})$ with $|T|\ge\ell$. This is the original
\path{A090-PARITY-TAIL-LOW-INFLUENCE-COMPLETENESS}, version 1
\cite{A090c}. A096 and A099 independently reconstruct this scoped theorem;
neither acceptance pays the resulting universal bill.
\begin{theorem}\label{thm:tail}
For every finite-valued nonnegative $s(N)=o(N)$ and fixed $\ell\ge2$,
the supremum of $R_{\ge\ell}(F)/I(F)$ over all actual Boolean $F$
with $0<I(F)\le2^{-s(N)}$ equals $K_E$, in the extended reals.
\end{theorem}
\begin{proof}
Take any nonconstant $n$-bit seed $f$, $m$ disjoint copies,
$h=\chi_Z f^{\otimes m}$ with $|Z|=\ell$ fresh parity coordinates,
and $F=1+A_k(h-1)$ with $k$ fresh point controls, $q=2^{-k}$.
The actual coefficients are $1-q$ at the empty frequency, $-q$ at
nonempty pure-control frequencies, and $q\widehat h(S)$ on every
control slice of each nonempty core frequency $S$.
All occupied core frequencies contain $Z$. Thus all singleton core
coefficients vanish, $B(h)=mB(f)$, and $I(h)=mI(f)+\ell$.
Every data edge with positive cost has both endpoints containing $Z$.
Homogeneity and the $2^k$ control slices give exactly $qmB(f)$,
entirely inside the retained tail. Each parity coordinate has zero erasure
but influence $q>0$; there are no two adjacent occupied frequencies on it.

For a control coordinate, nonempty core frequencies contribute $q$
to the erasure, since all such pairs have equal masses and their sum
of upper masses is $q/2$. Empty-core lower control sets of size $r$
give $\binom{k-1}{r}$ equal-mass pairs, each of cost $2q^2$.
The exceptional pair $((1-q)^2,q^2)$ is outside the retained tail.
Consequently
\begin{align}
I(F)&=q(mI(f)+\ell+k),\label{eq:tailI}\\
R_{\ge\ell}(F)&=qmB(f)+kq+
 2kq^2\sum_{r=\ell}^{k-1}\binom{k-1}{r},\label{eq:tailB}\\
R_{\ge2}(F)&=qmB(f)+2kq(1-kq).\label{eq:tailtwo}
\end{align}
An empty sum is zero. In particular the actual root and next layers are
$B_0=k\phi((1-q)^2,q^2)$ and $T_{12}=2k(k-1)q^2$.
No seed cost was removed by the fresh parity.

To verify the cutoff without assuming monotonicity, put
$\eta_m=\sup_{N\ge mn}s(N)/N$ and
$k_m=\lceil4(n+1)m\sqrt{\eta_m}+2\sqrt m\rceil$.
The tail supremum is finite and tends to zero, so $k_m=o(m)$.
Eventually $N_m=mn+\ell+k_m\le(n+2)m$ and $\eta_m\le1$.
Then $s(N_m)\le\eta_m(n+2)m\le4(n+1)m\sqrt{\eta_m}$.
The remaining $2\sqrt m$ dominates
$\log_2(mI(f)+\ell+k_m)\le\log_2((n+2)m)$.
Equation~\eqref{eq:tailI} therefore meets the cutoff. Equation~\eqref{eq:tailB}
gives a lower ratio tending to $B(f)/I(f)$, while
$R_{\ge\ell}(F)\le B(F)$ gives the upper supremum bound $K_E$.
Apply this to each finite seed ratio, including arbitrarily large ratios
if $K_E=\infty$. There is no exchange of an unknown supremum and a limit.
\end{proof}

\section{Complete conditional KL: what compression must retain}
\label{sec:compression}
For any deterministic quotient $C_i$ of a complete derivative row,
the KL chain rule gives the exact influence-weighted lost refund
\begin{equation}\label{eq:quotientloss}
 L_i(C_i)=d_i[D_2(\mu_i\Vert r_i)-D_2(C_i\mu_i\Vert C_i r_i)]
 =d_i\sum_c\mu_i(c)D_2(\mu_i(\cdot\mid c)\Vert r_i(\cdot\mid c)).
\end{equation}
The coarsened downward cost is $\mathcal L_i+L_i(C_i)$.
Marginalizing the complete reference preserves normalization but may
discard an unbounded conditional divergence. It is distinct from
conditioning the reference on derivative support, which also drops its
support mass and invalidates the complete-reference identity.

\begin{proposition}[A001 fixed-degree obstruction]\label{prop:degree}
Fix $L\ge0$ and keep each residual subset of size at most $L$ individually,
merging all larger subsets into one tail. For the one-point sign $f_n$,

\begin{align*}

 \mathcal L^{L}(G_n)/I(G_n)
 &\ge\lfloor n/2\rfloor-2-\beta(n-1)\longrightarrow\infty,\\
 \beta&=e\sum_{j=0}^L\binom{n-1}{j}.
\end{align*}
This is \path{A001-BOUNDED-DEGREE-REFUND-LOSS}, version 1~\cite{A001d},
assembled at author scope, not an FEI counterexample.
\end{proposition}
\begin{proof}

Every influence is $e$. Its row downward price is
$e-e^2+e^2\log_2(((1-e)^2+e^2)/e^2)$, giving ratio at most
$1+2(n-1)e\le2$. With $a=(1-e)^2$, $W=1-a=2e-e^2$, its entropy is
$-a\log_2 a+2(n-1)W\le2e/\ln2+4(n-1)e<4ne$.

There are no auxiliary parity factors or dummy coordinates.

each of mass $e$. Both row laws have the same independent spectator
factor. The individually kept low-atom derivative mass is at most $\beta$.
Every likelihood ratio is at most $1/e$, so their total signed KL
contribution is at most $\beta(n-1)$.
The reference tail contains the event that exactly one spectator block
has degree greater than $L$ and all other spectators are empty.
This has mass $(m-1)z a^{m-2}$ with
$z=e^2\sum_{j=L+1}^n\binom nj\ge e$ by binomial symmetry.
Since $me\le1/4$, $m-1\ge m/2$ and Bernoulli's inequality gives

including when the signed term is negative. Subtract these divergence

exponentially decaying polynomial, proving divergence.
Equation~\eqref{eq:quotientloss} shows that all the loss is the
within-tail conditional KL. Discarding only the independent spectators

\end{proof}

\section{A root remainder allowance is not unused child surplus}

\begin{lemma}[Conservative complete-reference transport]
For every finite Boolean sign $f$, let $P(S)=\widehat f(S)^2$ and
let $Q$ be a probability law on all subsets with $Q(S)>0$ whenever
$P(S)>0$. For a physical query $j$, set
\[
 m_T=P(T)+P(T+j),\qquad v_T=Q(T)+Q(T+j).
\]
Give both actual children the same reference $v$. Then
\[
 \CE(P,Q)=B_j(f)+\tfrac12\sum_{\epsilon=\pm1}
                  \CE(P_{f_\epsilon},v)+R_j,
 \qquad R_j=D_2(P\Vert Q)-D_2(m\Vert v)\ge0.
 \tag{NR1}
\]
Every finite outcome-adaptive tree, using only these inherited
references, therefore satisfies
\[
 \CE(P_f,Q)=\sum_{u\text{ internal}}w_u(B_{j_u}(f_u)+R_u)
                  +\sum_{\ell\text{ leaf}}w_\ell\CE(P_{f_\ell},Q_\ell),
 \qquad w_u=2^{-\operatorname{depth}(u)}.
 \tag{NR2}
\]
\end{lemma}
\begin{proof}
Write $a_T=\widehat f(T)$ and $b_T=\widehat f(T+j)$.
The actual child coefficients are $a_T+\epsilon b_T$, hence their
mean squared masses are $a_T^2+b_T^2=m_T$. Parseval for the actual
Boolean children proves their normalization. If $v_T=0$, both
parent coefficients and both child coefficients vanish, so every
displayed cost is finite. The reference $v$ has total mass one,
including all reference-only atoms.

For $m_T>0$, disintegrate $P(T,b)$ and $Q(T,b)$ into their
marginal and conditional laws. The conditional cross entropy is
$H(P(b\mid T))+D_2(P(b\mid T)\Vert Q(b\mid T))$.
Multiplication by $m_T$ and summation give (NR1), because the first
conditional entropy sum is $B_j$ and cross entropy is linear in
its true-law argument. Unoccupied fibers contribute zero.
The conditional divergence is nonnegative by
$\ln x\le x-1$ applied to $x=q_b/p_b$ on occupied atoms:
$\sum p_b\ln(p_b/q_b)\ge\sum_{p_b>0}(p_b-q_b)\ge0$.
This also covers reference mass at unoccupied conditional atoms.

Multiply (NR1) at each internal node by its physical reach
probability. Every child's weighted cost cancels once against
its parent's child term. Finite depth makes the cancellation exact,
without any limit exchange. This proves (NR2). Signed coefficients
are required to determine each child law, even though marginalizing
the reference itself needs no phase information.
\end{proof}

\section{The signed reset bill is the remaining account test}

bill
\[
 Z=\sum_\ell w_\ell\sum_S P_\ell(S)
                \log_2\frac{Q'_\ell(S)}{\widetilde Q_\ell(S)}.
 \tag{RS1}
\]
Unchanged terminals are included by taking
$\widetilde Q_\ell=Q'_\ell$. Then exactly
\[
 A-(d+b)-\sum_\ell w_\ell\CE(P_\ell,\widetilde Q_\ell)
       =U+r-Z.
 \tag{RS2}
\]
Indeed the new complete terminal cost is $T'+Z$, and substitution
of $T=b+r+T'$ proves the identity. A negative reset bill releases
capacity; a positive bill consumes it. This is bookkeeping, with
no assertion that the sign or magnitude is favorable. It keeps
every complementary terminal and every executed debit.

continuation has $Z=0$ and is therefore automatic once the root
account is paid. A construction that additionally charges the
channel term $J_j$ or a code for a reference label must include
that extra debit as well; (RS2) does not pay it implicitly.

\begin{lemma}[Comparison-chain identity for a subprobability]
\label{lem:subprob8}
Let $D$ be a nonnegative measure of mass at most one with
$\CE(P_f,D)<\infty$. A physical query of $j$ has frequency marginal
$p'=M_jP_f$ and comparison marginal $D'=M_jD$. Then
\[
\CE(P_f,D)-\tfrac12\sum_{b=0}^1\CE(P_{f_{j=b}},D')
=B_j(f)+\sum_Tp'(T)\KL(P_f(\cdot\mid T)\Vert D(\cdot\mid T)).
\]
All occupied-fiber conditional measures on the right are normalized
probabilities. In particular the rightmost term is nonnegative, even
though $D$ need not be normalized globally.
\end{lemma}
\begin{proof}
In each occupied fiber factor $D(T,b)=D'(T)D(b\mid T)$ and
$P_f(T,b)=p'(T)P_f(b\mid T)$. Subtracting the two cross entropies cancels
the marginal term because the average child spectrum is $p'$.
The remaining conditional cross entropy is its conditional entropy
$B_j(f)$ plus its conditional KL. Finite root CE propagates as above,
so every occupied denominator is positive. Iteration on a finite
adaptive tree with actual reach $\pi_v$ yields
\[
\sum_{v\text{ internal}}\pi_v B_{j_v}(f_v)
+\sum_{\ell\text{ terminal}}\pi_\ell\CE(P_{f_\ell},M_{A_\ell}D)
\le\CE(P_f,D).
\]
No independence of branches is assumed and $D$ is never a terminal
reference actually supplied to the algorithm.
\end{proof}

\begin{theorem}[Supplied likelihood transfer at a first-reset frontier]

\]
A sharper excess is $\log_2(1/\gamma)$ times

\bigl[(1-m)+m\operatorname{Var}(g_\ell)\bigr].
\]
In particular $m^2=(\E g)^4$ in the displayed bound.
\end{theorem}
\begin{proof}

\end{proof}

$D(P\Vert Q)=\operatorname{CE}(P,Q)-H(P)$. Terms of true mass zero contribute zero.
References are complete probability laws, including their mass outside the true support.

\begin{lemma}[Sharp signed admission from the current reference]
\label{lem:v11-floor}
Let $p(0)=q_0(0)=r$, $t=1-r>0$, and let $A$ be a set of nonzero atoms.
Suppose $0<a=p(A)<t$ and $0<b=q_0(A)<\ell<t$, with finite cross entropy.
Admissible references keep the conditional laws of $q_0$ on $A$ and on its
nonzero complement, keep mass $r$ at zero, and have mass $z\geq\ell$ on $A$.
The unique minimizing mass is $z_* =\max(a,\ell)$, and the minimum signed change is
\[
 F(z_*)=a\log\frac b{z_*}+(t-a)\log\frac{t-b}{t-z_*}
 =t\left[D(\operatorname{Ber}(a/t)\Vert\operatorname{Ber}(z_*/t))
 -D(\operatorname{Ber}(a/t)\Vert\operatorname{Ber}(b/t))\right].
\]
\end{lemma}
\begin{proof}
Within each cell the likelihood ratio is constant, so direct expansion of cross entropy
gives $F(z)$ as displayed. Both cells have positive true mass; $z=t$ has infinite cost.
For $\ell\leq z<t$,
\[
 F'(z)=\frac{-a/z+(t-a)/(t-z)}{\ln2}
       =\frac{t(z-a)}{z(t-z)\ln2},\qquad
 F''(z)=\frac{a/z^2+(t-a)/(t-z)^2}{\ln2}>0.
\]
Every denominator is positive. The derivative changes sign at $a$, which proves the
minimum and uniqueness. Expanding the two Bernoulli divergences gives the second formula.
The formula is signed: moving a mismatched incoming reference toward the truth can release
previously paid cost. True-zero reference atoms remain in their conditional profiles.
\end{proof}

If full terminal laws have an identical disjoint controller part and a seed part of mass
$\beta_h$, the full change is $\beta_h F_h(z_{*,h})$. At actual physical reaches $\rho_h$,
let $C$ be the sum of all current full terminal costs, $U$ all earlier outstanding bills,
and $B_0$ the original supplied capacity. Exact affordability is
\[
 \sum_h\rho_h\beta_h F_h(z_{*,h})\leq B_0-C-U.
\]
This follows by subtracting the old cost from the new cost, retaining both outcomes.
A negative change is released once; it is not an additional independent refund.
The condition assumes this particular class is admissible. It does not prove that any
useful algorithm requires the class or that $B_0=O(I)$ exists for arbitrary cores.

\begin{lemma}[Ghost-complete inherited witness transport]
\label{lem:v11-ghost}
Let $P$ be the squared Fourier law of a Boolean function on a finite uniform cube and
\[
 Q=bP+\sum_{j=1}^k c_j\delta_{W_j},\qquad
 b>0,\quad c_j\geq0,\quad b+\sum_jc_j=1.
\]
Witnesses may be repeated or have zero true mass. Query a physical coordinate $i$.
Let $M$ delete its frequency label, $R=MP$, $\overline Q=MQ$, and let $P_0,P_1$
be the actual child spectra. On rows with $R(u)>0$, set $K_t(u)=P_t(u)/(2R(u))$.
On rows with $R(u)=0$, choose any stochastic pair $K_0(u)+K_1(u)=1$.
Put $r_t=\sum_u\overline Q(u)K_t(u)$. Then $r_t\geq b/2>0$ and the full child reference is
\[
 Q_t=\frac{K_t\overline Q}{r_t}
 =\frac b{2r_t}P_t+
 \sum_j\frac{c_jK_t(MW_j)}{r_t}\delta_{MW_j}.
\]
Thus the original $k$ labelled witnesses suffice at every finite adaptive descendant.
\end{lemma}
\begin{proof}
For exterior Fourier coefficients $u,v$ of the parent, the two restriction coefficients
are $u+v$ and $u-v$. Hence $(P_0+P_1)/2=R$, including zero rows, and $RK_t=P_t/2$.
Marginalizing the reference and multiplying by this same kernel gives the formula.
Summing it gives $r_t=b/2+\sum_jc_jK_t(MW_j)$. Consequently the child weights are
nonnegative and sum to one. The choice on zero rows carries reference-only mass; it
does not affect the true joint law. Deleting labels preserves their provenance even
when labels collide, acquire true mass, or lose their weight. Induction proves the
finite-tree assertion, with the kernel chosen from the actual spectra at each node.
\end{proof}

\begin{lemma}[Complete accounting under the same stochastic kernel]
\label{lem:v11-account}
In Lemma~\ref{lem:v11-ghost}, write $A=D(P\Vert Q)$,
$A_t=D(P_t\Vert Q_t)$, and
\[
 D_{\rm erase}=D(P\Vert Q)-D(R\Vert\overline Q),\quad
 D_{\rm out}=D((1/2,1/2)\Vert(r_0,r_1)),\quad
 J=H(R)-\tfrac12\sum_tH(P_t),\quad B_i=H(P)-H(R).
\]
All four quantities $D_{\rm erase},D_{\rm out},J,B_i$ are nonnegative, and
\[
 A=D_{\rm erase}+D_{\rm out}+\tfrac12\sum_t A_t,
\]
\[
 B_i+\tfrac12\sum_t\operatorname{CE}(P_t,Q_t)-\operatorname{CE}(P,Q)
 =-J-D_{\rm erase}-D_{\rm out}.
\]
Also $\tfrac12\sum_t\log(1/b_t)=\log(1/b)-D_{\rm out}$.
Every finite adaptive tree telescopes these equalities at its actual reaches, retaining
all full terminal costs and the original reference account.
\end{lemma}
\begin{proof}
The conditional chain rule after deleting a frequency gives
$D_{\rm erase}=\sum_uR(u)D(P(i\mid u)\Vert Q(i\mid u))\geq0$.
Applying the same stochastic kernel to $R$ and $\overline Q$ gives joint laws
$P_t(u)/2$ and $\overline Q(u)K_t(u)$. Their divergence is $D(R\Vert\overline Q)$:
on positive true rows the kernel cancels; on zero true rows the summands vanish,
but reference mass remains in $r_t$. The joint chain rule by outcome gives
$D(R\Vert\overline Q)=D_{\rm out}+\sum_t A_t/2$.
Entropy concavity gives $J\geq0$, and $B_i$ is conditional occupancy entropy.
Since $H(P)-\sum_tH(P_t)/2=B_i+J$, subtracting the entropy difference from the
divergence difference gives the full cross-entropy equality. Finally
$b_t=b/(2r_t)$ gives the stated logarithmic identity by direct expansion.
Multiplication by the reach of each internal node cancels every child account against
its parent account. Physical outcomes have conditional reach $1/2$, irrespective of
the reference normalizers $r_t$. This proves the stopped-tree identities.
\end{proof}

Any normalized finite-CE reference $Q$ admits the mixture in
Lemma~\ref{lem:v11-ghost}: choose
$0<b<\min(1,\min_{P(s)>0}Q(s)/P(s))$ and decompose the nonnegative law $Q-bP$
into point masses. This can require $2^n$ labels and says nothing about the initial
cross entropy. A supplied compact mixture remains compact; universal compact pricing
does not follow.

\paragraph{Distinct inherited-reference policies.}
If both children instead use the common marginal $\overline Q$, the correct identity is
\[
 B_i+\tfrac12\sum_t\operatorname{CE}(P_t,\overline Q)
 -\operatorname{CE}(P,Q)=-D_{\rm erase}.
\]
It follows by the conditional cross-entropy chain rule. This is a different policy
from the outcome-adapted references $Q_t$ above; the latter release the additional
$J+D_{\rm out}$. Neither policy creates a new initial budget. For the sign table
$(-1,1,1,1)$ with $Q=P$, the common-marginal cost change is $-1$, while
$D_{\rm erase}=0$ and $B_i=1$. Omitting $B_i$ from the equality is false.

\paragraph{Signed Boolean control.}
For $g=(1-y_1y_2-y_1y_3-y_2y_3)/2$, either child after querying $y_1$ has
spectral law $p=(1/4,1/4,1/4,1/4)$ on the exterior labels.
Let $A$ be the last label and $\ell=1/3$. The two incoming references
$p$ and $(1/4,5/16,5/16,1/8)$ have the same optimal target
$(1/4,5/24,5/24,1/3)$. Their signed costs are respectively
$\frac14\log(27/25)>0$ and $\frac14\log(27/32)<0$.
The sign difference is exactly a difference of prepaid mismatch.

\section{Full physical-query bills and the price of changing reference policy}
\label{sec:v12}
All logarithms are base two. The reference law includes every atom outside the true
support. An old bill is retained as a fixed term $U$ throughout.

\begin{lemma}[Exact projection-query bill]\label{lem:v12-query}
Let $P$ be a Boolean squared Fourier law and let $Q$ be a normalized covering reference.
At a physical query $i$, give both children $Q'=\pi_iQ$, where $\pi_i$ deletes frequency $i$.
Put $P'=\pi_iP$, $L_i=D(P\Vert Q)-D(P'\Vert Q')$ and
$J_i=\frac12\sum_{b=0}^1D(P_b\Vert P')$. Then
\[
 L_i\geq0,\quad 0\leq J_i\leq\frac2{\ln2}\operatorname{Inf}_i(f),\quad
 \tfrac12\sum_bD(P_b\Vert Q')=D(P\Vert Q)+J_i-L_i.
\]
\end{lemma}
\begin{proof}
The restriction coefficients are $a_T,b_T$; the parent coefficients are
$(a_T+b_T)/2,(a_T-b_T)/2$. Squaring gives
$P'=(P_0+P_1)/2$. The KL chain rule along $\pi_i$ gives $L_i\geq0$; expanding the
child divergences about their midpoint gives the equality. Every positive child atom
is covered by $Q'$, so all divergences are finite.

For completeness, in natural units define
\[
 e(a,b)=\tfrac12\left[a^2\ln\frac{2a^2}{a^2+b^2}
                       +b^2\ln\frac{2b^2}{a^2+b^2}\right].
\]
We have $e(a,b)\leq(a-b)^2/2$. Taking absolute values leaves the left side unchanged
and only decreases the right side. For nonnegative $a,b$ with $a+b>0$, scale to
$a=1+t,b=1-t$, $|t|\leq1$. If $j(t)=e(1+t,1-t)$, then
\[
 j(0)=j'(0)=0,\qquad
 j''(t)=2\ln\frac{1-t^2}{1+t^2}+\frac4{1+t^2}\leq4\quad(-1<t<1).
\]
Integrating twice and using evenness gives $j(t)\leq2t^2$. Continuity treats the
endpoints; $a=b=0$ contributes zero. Summation over $T$ and division by $\ln2$ gives
the asserted bound because $\operatorname{Inf}_i=\sum_T(a_T-b_T)^2/4$.
\end{proof}

For any covering update $Q\longmapsto R$ with finite forward fee, define its excess
\[
 G(P;Q,R)=D(R\Vert Q)+D(P\Vert R)-D(P\Vert Q)
          =\sum_s(R(s)-P(s))\log\frac{R(s)}{Q(s)}.
\]
Terms with $R(s)=P(s)=0$ are zero. Updates that fail these support conditions cannot
be inserted as finite-fee transactions.

\begin{lemma}[Two updates with nonpositive excess]\label{lem:v12-update}
For a finite partition with true and reference cell masses $p,q$, replace the reference
cell masses by $r=(1-t)q+tp$, $0\leq t\leq1$, preserving its conditional laws in the cells.
The complete new reference $R$ satisfies $G\leq0$. For $0<t<1$,
\[
 G=-\frac{1-t}{t}\{D(r\Vert q)+D(q\Vert r)\}.
\]
At $t=0$ and $t=1$, $G=0$.

Alternatively, let $R$ minimize $D(v\Vert Q)$ over a nonempty closed affine slice of
the simplex containing $P$, with support restricted to $\operatorname{supp}Q$.
Then $R$ covers $P$ and $G=0$.
\end{lemma}
\begin{proof}
For the partition update, conditionals cancel, and
$G=\sum_c(r_c-p_c)\log(r_c/q_c)$.
For $0<t<1$, $r-p=-(1-t)(r-q)/t$, giving the formula. A zero $q$ cell has zero
$p$ and zero $r$ and is omitted jointly. At $t=0$ the update is the identity.
At $t=1$, $r=p$ on the cells, so $G=0$, including cells whose reference mass vanishes.

For the affine update the feasible set is compact and the objective continuous and
strictly convex, giving a unique minimizer. It contains the support of every feasible
law: otherwise mixing it toward that law creates a negative $t\log(1/t)$ term, while
changes on its positive support are $O(t)$, contradicting minimality. Finiteness of the
set justifies this comparison. In particular it covers $P$.
Both signs of the direction $P-R$ are feasible for sufficiently small displacement:
outside $\operatorname{supp}R$ both laws vanish, and on its finite positive support the
entries remain nonnegative. First variation therefore gives
$\sum_s(P(s)-R(s))\log(R(s)/Q(s))=0$, which is $G=0$.
\end{proof}

\begin{theorem}[One supplied source pays complete projection histories]\label{thm:v12-bill}
Let $f$ be any Boolean function on a finite uniform cube, and supply a normalized
covering reference $Q_0$. Consider any finite adaptive tree with physical coordinate
queries, projection inheritance as in Lemma~\ref{lem:v12-query}, and finitely many
updates of Lemma~\ref{lem:v12-update}. Let $w_v$ be actual physical reaches.
Charge signed physical entropy increments, full query contractions $L_v$, every full
forward update fee, and full terminal cross entropy. Then the complete bill obeys
\[
 \mathrm{Bill}=U+\operatorname{CE}(P_0,Q_0)
               +\sum_{\rm query}w_vJ_v+\sum_{\rm update}w_vG_v
 \leq U+\operatorname{CE}(P_0,Q_0)
     +\frac2{\ln2}\left(I(f)-\mathbb E I(f_{\rm leaf})\right).
\]
Each physical coordinate is queried at most once per path. There is no new source
allowance at any descendant. A fixed positive subprobability scalar can also be
preserved, with its cross-entropy penalty retained.
\end{theorem}
\begin{proof}
Lemma~\ref{lem:v12-query} gives the change in true divergence at each query.
At an update its change plus the forward fee is $G$.
Multiplication by actual reaches and cancellation of all interior divergences gives
\[
 \mathbb E D_{\rm leaf}+\sum_{\rm query}w_vL_v
   +\sum_{\rm update}w_v\mathrm{fee}_v
 =D_0+\sum_{\rm query}w_vJ_v+\sum_{\rm update}w_vG_v.
\]
Signed entropy changes telescope to $H(f)-\mathbb EH(f_{\rm leaf})$, proving the
exact bill. Updates change only references, so
$\sum_{\rm query}w_v\operatorname{Inf}_{i_v}(f_v)
=I(f)-\mathbb EI(f_{\rm leaf})$.
The two lemmas now give the inequality. All sums are finite; no optional-stopping
limit is invoked. For a fixed scalar $s\in(0,1]$, replacing every reference by $sQ$
adds $-\log s$ to its true cross entropy and divergence, unchanged along a query;
the update fee is the divergence between normalized shapes. The same proof retains
that scalar penalty.
\end{proof}

This theorem does not permit an arbitrary update at zero excess. The direction
$D(R\Vert Q)$ of the forward fee matters, as does the specified inheritance policy.

\begin{proposition}[A positive forward handoff on an actual rate-four source]
\label{prop:v12-forward}
Let $f$ have one negative vertex on three bits, and supply the uniform reference on
eight spectral atoms. Then $I(f)=3/4$ and $\operatorname{CE}=3=4I(f)$.
After any physical root query, switching from the projected reference to the normalized
true-kernel reference has average excess $\frac3{32}\log5>0$ over both children.
\end{proposition}
\begin{proof}
The parent spectrum is $(9/16,1/16,\ldots,1/16)$.
The true exterior midpoint and projected reference are
$m=(5/8,1/8,1/8,1/8)$ and $q=(1/4,1/4,1/4,1/4)$.
The children are $P_c=\delta_0$ and $P_a=q$. Normalize
$R_b(T)=q(T)P_b(T)/(m(T)Z_b)$.
Direct division gives $Z_c=2/5$, $R_c=\delta_0$, $Z_a=8/5$, and
$R_a=(1/16,5/16,5/16,5/16)$. All forward fees are finite.
For the constant child the excess is zero. Since $P_a=q$, the other excess is
\[
 D(R_a\Vert P_a)+D(P_a\Vert R_a)
 =\left(\tfrac{15}{16}\log5-2\right)
  +\left(2-\tfrac34\log5\right)=\tfrac3{16}\log5.
\]
Each child has actual reach $1/2$. Translation of inputs and output complementation
preserve the squared laws, proving the full stated symmetry scope.
\end{proof}

\begin{proposition}[High-order matching does not pay a query locally]
\label{prop:v12-moment}
For every even $n\geq4$, an actual Boolean function has $H=n$, $I=n/2$ and a positive
rational reference of cross entropy below $4I$ matching the vacuum and all spectral
marginals on at most $n-2$ coordinates, but each $i\leq n-1$ has
\[
 n-\tfrac14<L_i<n,\qquad 2n-\tfrac12<L_i/\operatorname{Inf}_i<2n.
\]
\end{proposition}
\begin{proof}
Take the product of $n/2$ disjoint two-bit signs with one negative input each.
Their squared spectra are uniform, so $P(S)=2^{-n}$ and every influence is $1/2$.
Put $k=n-1$, $\epsilon=2^{-4n}$, and perturb the face
$F=\{\{n\}\cup V:V\subseteq[k]\}$ by
\[
 Q(\{n\}\cup V)=2^{-n}\{1+(1-\epsilon)(-1)^{|V|}\},
 \qquad Q(S)=2^{-n}\quad(S\notin F).
\]
Normalization follows by parity cancellation; all entries are positive and the
vacuum lies outside the face. Every inclusion moment of order at most $k-1$ has an
unsummed free coordinate on the face, so its perturbation vanishes.
Inclusion-exclusion gives all marginals of that order. For $i\leq k$ the perturbations
cancel within each projection fiber; consequently $\pi_iQ=\pi_iP$ and
\[
 L_i=D(P\Vert Q)=\tfrac14\log\frac1{\epsilon(2-\epsilon)}
                =n-\tfrac14\log(2-\epsilon).
\]
The strict bounds follow from $1<2-\epsilon<2$. Thus $\operatorname{CE}=n+L_i<2n=4I$.
Any information projection imposing only these matched moments leaves $Q$ fixed:
it is feasible and has zero divergence from itself. These are alternative first
queries, not $n-1$ separate debits on one path. After the first such projection the
original mismatch is gone. No repeated unused reserve is inferred.
\end{proof}

\paragraph{Initial source pricing remains universal.}
Let $g$ be balanced and let a fresh $r$-bit point event have probability $p=2^{-r}$.
For $F=1-A+Ag$, direct Fourier expansion gives
\[
 H(F)=A_r+pH(g),\quad I(F)=p(I(g)+r),\quad
 A_r=2rp(2-p)-(1-p)^2\log((1-p)^2).
\]
For any covering full reference $Q_F$, its nonempty-data sector has positive mass
$q_D$. Marginalize its conditional reference over the control frequencies to obtain
a covering law $R_g$ on nonempty data sets. The true sector mass is $p$, its
conditional law is uniform on control frequencies times $P_g$, and the KL chain rule
followed by marginalization gives $D(P_F\Vert Q_F)\geq pD(P_g\Vert R_g)$.
Therefore
$\operatorname{CE}(P_F,Q_F)\geq A_r+p\operatorname{CE}(P_g,R_g)$.
For arbitrary $h$, choose $g=z\prod_{j=1}^m h_j$ on disjoint variables. Then
$H(g)=mH(h)$ and $I(g)=1+mI(h)$.
A universal original source bound $\operatorname{CE}(P_F,Q_F)\leq KI(F)$ would imply
\[
 H(h)\leq KI(h)+\frac{K(1+r)-A_r/p}{m}.
\]
Letting integers $m$ tend to infinity proves full FEI with constant $K$.
History stability cannot silently replace this unpaid initial-source theorem.
